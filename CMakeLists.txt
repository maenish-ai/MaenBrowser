cmake_minimum_required(VERSION 3.21)
project(MaenBrowser VERSION 1.6.7 LANGUAGES CXX RC)
if(NOT WIN32)
 message(FATAL_ERROR "Windows only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(WEBVIEW2_SDK_ROOT "" CACHE PATH "Microsoft.Web.WebView2 NuGet extraction root")
if(NOT EXISTS "${WEBVIEW2_SDK_ROOT}/build/native/include/WebView2.h")
 message(FATAL_ERROR "WEBVIEW2_SDK_ROOT is required")
endif()
add_executable(MaenBrowser WIN32 src/webview2_browser/main.cpp src/win/maenbrowser.rc)
target_include_directories(MaenBrowser PRIVATE "${WEBVIEW2_SDK_ROOT}/build/native/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(MaenBrowser PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(MaenBrowser PRIVATE "${WEBVIEW2_SDK_ROOT}/build/native/x64/WebView2LoaderStatic.lib" user32 gdi32 shell32 ole32 oleaut32 advapi32 version shlwapi runtimeobject)
